cmake_minimum_required(VERSION 3.16)
project(obs-ltc-source VERSION 1.0.0)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

# The plugin is dlopen()'d by a host OBS process that has already loaded its
# own libobs by soname; it must resolve against that instance, not risk
# mapping a second, mismatched copy via an rpath baked in at our build time.
set(CMAKE_SKIP_BUILD_RPATH TRUE)
set(CMAKE_SKIP_INSTALL_RPATH TRUE)

# --- libobs ------------------------------------------------------------
# If OBS Studio was installed system-wide with CMake package config exported
# (e.g. built with `cmake --install`), this picks it up automatically.
find_package(libobs QUIET)

if(NOT TARGET OBS::libobs)
    # Otherwise, point CMake at an existing obs-studio checkout/build, e.g.:
    #   cmake -B build \
    #     -DLIBOBS_INCLUDE_DIR=/path/to/obs-studio/libobs \
    #     -DLIBOBS_LIB=/path/to/obs-studio/build/libobs/libobs.so
    #
    # obsconfig.h is generated by OBS's own CMake configure step and lands in
    # its build tree, not the libobs/ source dir, so it needs its own include
    # path. Defaults to the standard <build>/config location relative to
    # LIBOBS_LIB; override with -DLIBOBS_CONFIG_INCLUDE_DIR if yours differs.
    if(NOT LIBOBS_INCLUDE_DIR OR NOT LIBOBS_LIB)
        message(FATAL_ERROR
            "libobs was not found via find_package(). Build or install OBS Studio "
            "(https://github.com/obsproject/obs-studio) then re-run CMake with:\n"
            "  -DLIBOBS_INCLUDE_DIR=/path/to/obs-studio/libobs\n"
            "  -DLIBOBS_LIB=/path/to/obs-studio/build/libobs/libobs.so")
    endif()
    if(NOT LIBOBS_CONFIG_INCLUDE_DIR)
        get_filename_component(_libobs_lib_dir "${LIBOBS_LIB}" DIRECTORY)
        set(LIBOBS_CONFIG_INCLUDE_DIR "${_libobs_lib_dir}/../config")
    endif()
    add_library(OBS::libobs SHARED IMPORTED)
    if(WIN32)
        # On Windows, LIBOBS_LIB is the import library (obs.lib); the linker
        # needs that as IMPORTED_IMPLIB, while IMPORTED_LOCATION is the
        # actual runtime obs.dll (defaults to the same directory, matching
        # a normal obs-studio build layout; override with -DLIBOBS_DLL).
        if(NOT LIBOBS_DLL)
            get_filename_component(_libobs_lib_name "${LIBOBS_LIB}" NAME_WE)
            get_filename_component(_libobs_lib_dir "${LIBOBS_LIB}" DIRECTORY)
            set(LIBOBS_DLL "${_libobs_lib_dir}/${_libobs_lib_name}.dll")
        endif()
        set_target_properties(OBS::libobs PROPERTIES
            IMPORTED_IMPLIB "${LIBOBS_LIB}"
            IMPORTED_LOCATION "${LIBOBS_DLL}"
            INTERFACE_INCLUDE_DIRECTORIES "${LIBOBS_INCLUDE_DIR};${LIBOBS_CONFIG_INCLUDE_DIR}")
    else()
        set_target_properties(OBS::libobs PROPERTIES
            IMPORTED_LOCATION "${LIBOBS_LIB}"
            INTERFACE_INCLUDE_DIRECTORIES "${LIBOBS_INCLUDE_DIR};${LIBOBS_CONFIG_INCLUDE_DIR}")
    endif()
endif()

# --- Freetype2 (cross-platform text rasterization) ----------------------
find_package(Freetype REQUIRED)

# --- Fontconfig (Linux family/style -> font file resolution) ------------
if(UNIX AND NOT APPLE)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(FONTCONFIG REQUIRED fontconfig)
endif()

# --- libltc (LTC decode) -------------------------------------------------
# Vendored directly (see third_party/libltc/COPYING.libltc, LGPL-3.0+) rather
# than requiring a system package, so the plugin builds the same way on
# Linux/Windows/macOS without extra install steps.
add_library(ltc_vendored STATIC
    third_party/libltc/decoder.c
    third_party/libltc/encoder.c
    third_party/libltc/ltc.c
    third_party/libltc/timecode.c
)
target_include_directories(ltc_vendored PUBLIC third_party/libltc)
set_target_properties(ltc_vendored PROPERTIES POSITION_INDEPENDENT_CODE ON)

add_library(obs-ltc-source MODULE
    src/plugin-main.cpp
    src/ltc-source.cpp
    src/ltc-source.h
    src/font-resolver.cpp
    src/font-resolver.h
    src/frontend-recording.cpp
    src/frontend-recording.h
)

target_link_libraries(obs-ltc-source PRIVATE OBS::libobs Freetype::Freetype ltc_vendored)

if(UNIX)
    # frontend-recording.cpp resolves obs-frontend-api symbols at runtime via
    # dlopen()/dlsym() rather than linking against it at build time (see that
    # file for why).
    target_link_libraries(obs-ltc-source PRIVATE ${CMAKE_DL_LIBS})
endif()

if(UNIX AND NOT APPLE)
    target_include_directories(obs-ltc-source PRIVATE ${FONTCONFIG_INCLUDE_DIRS})
    target_link_libraries(obs-ltc-source PRIVATE ${FONTCONFIG_LIBRARIES})
    target_compile_definitions(obs-ltc-source PRIVATE HAVE_FONTCONFIG)
endif()

set_target_properties(obs-ltc-source PROPERTIES PREFIX "")

# Ship data/ (locale strings) as a sibling of the built module, for
# convenient local testing (point OBS's plugin data dir at this).
add_custom_command(TARGET obs-ltc-source POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory
        "${CMAKE_CURRENT_SOURCE_DIR}/data"
        "$<TARGET_FILE_DIR:obs-ltc-source>/data"
)
